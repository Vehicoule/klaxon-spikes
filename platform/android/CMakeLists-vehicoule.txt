cmake_minimum_required(VERSION 3.22)

# CMakeLists-vehicoule.txt — cible APK Vehicoule (V1 poche).
# Renommer en CMakeLists.txt dans app/jni/src/ (remplace la variante gallery).
# Staging attendu sous app/jni/src/ :
#   veh/        = copie du repo vehicoule/ (decoder.c + vendor/ + DejaVuSans.ttf)
#   zig/vehicoule/main.zig, zig/klaxon/klaxon.zig, ph/runtime.zig
#   kx_gallery_glue.cpp, kx/kx_skia_android.cpp, kx/kx_draw.cpp, kx/kx_scenes.cpp
#   vehicoule.o (zig build-obj — produit par scripts/build_veh.sh)
project(kxveh CXX C)

if(NOT TARGET SDL3::SDL3)
    find_package(SDL3 CONFIG)
endif()

if(NOT TARGET SDL3::SDL3)
    find_library(SDL3_LIBRARY NAMES "SDL3")
    find_path(SDL3_INCLUDE_DIR NAMES "SDL3/SDL.h")
    add_library(SDL3::SDL3 UNKNOWN IMPORTED)
    set_property(TARGET SDL3::SDL3 PROPERTY IMPORTED_LOCATION "${SDL3_LIBRARY}")
    set_property(TARGET SDL3::SDL3 PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${SDL3_INCLUDE_DIR}")
endif()

if(NOT TARGET SDL3::SDL3)
    message(FATAL_ERROR "Cannot find SDL3.")
endif()

# ---- libs Skia statiques (produites par k0-android/scripts/build_skia.sh) ----
set(KX_SKIA_OUT "" CACHE PATH "répertoire out/android-<arch> du build skia")
if(NOT KX_SKIA_OUT)
    set(KX_SKIA_OUT "$ENV{KX_SKIA_OUT}")
endif()
if(NOT KX_SKIA_OUT)
    message(FATAL_ERROR "KX_SKIA_OUT manquant (ex. ~/kx/deps/skia/out/android-x64)")
endif()
# per-ABI : arm64-v8a→out/android-arm64, x86_64→out/android-x64
# (KX_SKIA_OUT = base, on dérive le sibling par ABI)
if(CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
    get_filename_component(KX_SKIA_OUT "${KX_SKIA_OUT}/../android-arm64" ABSOLUTE)
else()
    get_filename_component(KX_SKIA_OUT "${KX_SKIA_OUT}/../android-x64" ABSOLUTE)
endif()
if(NOT EXISTS "${KX_SKIA_OUT}/libskia.a")
    message(FATAL_ERROR "libskia.a absent pour ${CMAKE_ANDROID_ARCH_ABI} : ${KX_SKIA_OUT}")
endif()
get_filename_component(KX_SKIA_DIR "${KX_SKIA_OUT}/../.." ABSOLUTE)

set(KX_SKIA_LIBS
    skia skparagraph skshaper skunicode_core skunicode_icu
    harfbuzz freetype2 png jpeg jpeg12 jpeg16 skcms raw_ptr
    cpu-features expat allocator_base allocator_core)

set(KX_SKIA_LIB_TARGETS "")
foreach(name IN LISTS KX_SKIA_LIBS)
    if(EXISTS "${KX_SKIA_OUT}/lib${name}.a")
        add_library(kx_${name} STATIC IMPORTED GLOBAL)
        set_property(TARGET kx_${name} PROPERTY IMPORTED_LOCATION
                     "${KX_SKIA_OUT}/lib${name}.a")
        list(APPEND KX_SKIA_LIB_TARGETS kx_${name})
    endif()
endforeach()

# vehicoule-<abi>.o : objet zig 0.17 pré-compilé par scripts/build_veh.sh
if(CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
    set(VEH_OBJ "${CMAKE_CURRENT_SOURCE_DIR}/vehicoule-arm64.o")
else()
    set(VEH_OBJ "${CMAKE_CURRENT_SOURCE_DIR}/vehicoule-x64.o")
endif()
if(NOT EXISTS "${VEH_OBJ}")
    message(FATAL_ERROR "${VEH_OBJ} manquant — lancer scripts/build_veh.sh (zig build-obj)")
endif()

# ---- décodeurs vendored (pur C) : dr_libs + stb_vorbis + opusfile/libopus/libogg ----
set(V "${CMAKE_CURRENT_SOURCE_DIR}/veh/vendor")
file(GLOB VEH_CELT    "${V}/opus/celt/*.c")
file(GLOB VEH_SILK    "${V}/opus/silk/*.c")
file(GLOB VEH_SILKF   "${V}/opus/silk/float/*.c")
file(GLOB VEH_OPUS    "${V}/opus/src/*.c")
set(VEH_DEC
    veh/decoder.c
    ${V}/stb_vorbis.c
    ${V}/ogg/src/bitwise.c
    ${V}/ogg/src/framing.c
    ${V}/opusfile/src/opusfile.c
    ${V}/opusfile/src/info.c
    ${V}/opusfile/src/internal.c
    ${V}/opusfile/src/stream.c
    ${VEH_CELT} ${VEH_SILK} ${VEH_SILKF} ${VEH_OPUS})
list(FILTER VEH_DEC EXCLUDE REGEX "(_demo|opus_compare)\\.(c|o)$")

add_library(main SHARED
    kx_gallery_glue.cpp
    kx/kx_skia_android.cpp
    kx/kx_scenes.cpp
    kx/kx_draw.cpp
    ${VEH_DEC}
    "${VEH_OBJ}"
)
set_source_files_properties("${VEH_OBJ}" PROPERTIES EXTERNAL_OBJECT true GENERATED true)

set_source_files_properties(kx/kx_skia_android.cpp kx/kx_scenes.cpp kx/kx_draw.cpp
    PROPERTIES COMPILE_OPTIONS "-fno-exceptions;-fno-rtti")

target_include_directories(main PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/veh
    ${V}/opus/include ${V}/opus/celt ${V}/opus/silk ${V}/opus/silk/float ${V}/opus/src
    ${V}/ogg/include ${V}/opusfile/include
    ${KX_SKIA_DIR}
    ${KX_SKIA_DIR}/third_party/externals/vulkan-headers/include
    ${KX_SKIA_DIR}/third_party/externals/vulkanmemoryallocator/include)

target_compile_features(main PRIVATE cxx_std_20)
target_compile_definitions(main PRIVATE
    SK_GANESH SK_GRAPHITE SK_USE_PERFETTO=0 VK_USE_PLATFORM_ANDROID_KHR
    OPUS_BUILD USE_ALLOCA HAVE_ALLOCA_H HAVE_LRINT HAVE_LRINTF HAVE_LROUND VAR_ARRAYS
    KX_MAIN_SYM=kx_vehicoule_main)
target_compile_options(main PRIVATE -fvisibility=hidden -Oz
    $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions -fno-rtti -fno-unwind-tables>
    $<$<COMPILE_LANGUAGE:C>:-fno-unwind-tables>)
set_property(TARGET main PROPERTY LINK_DEPENDS "${VEH_OBJ}")

target_link_libraries(main PRIVATE
    SDL3::SDL3
    -Wl,--start-group ${KX_SKIA_LIB_TARGETS} -Wl,--end-group
    EGL GLESv3 log dl m z android icu jnigraphics
    -Wl,-u,kx_vehicoule_main)

# 16KiB page-size (API35+) — alignement ELF 16K. gc-sections+icf :
# libmain 12,8→8,59 Mo mesuré (BUILD-PROFILE.md).
target_link_options(main PRIVATE
    -Wl,-z,max-page-size=16384
    -Wl,--gc-sections
    -Wl,--icf=all)
